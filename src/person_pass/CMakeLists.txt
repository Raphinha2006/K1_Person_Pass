cmake_minimum_required(VERSION 3.16)
project(person_pass)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(booster_interface REQUIRED)
find_package(booster_msgs REQUIRED)
find_package(vision_interface REQUIRED)
include(CTest)
add_executable(person_pass_node src/person_pass_node.cpp)
target_compile_features(person_pass_node PRIVATE cxx_std_17)
target_compile_options(person_pass_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(person_pass_node PRIVATE include)
ament_target_dependencies(person_pass_node rclcpp std_msgs std_srvs booster_interface booster_msgs vision_interface)
install(TARGETS person_pass_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(controller_test test/controller_test.cpp)
  target_compile_features(controller_test PRIVATE cxx_std_17)
  target_include_directories(controller_test PRIVATE include)
  add_test(NAME controller_test COMMAND controller_test)
  add_executable(adapter_test test/adapter_test.cpp)
  target_compile_features(adapter_test PRIVATE cxx_std_17)
  ament_target_dependencies(adapter_test rclcpp std_msgs booster_interface booster_msgs vision_interface)
  add_dependencies(adapter_test person_pass_node)
  add_test(NAME adapter_test COMMAND adapter_test $<TARGET_FILE:person_pass_node>)
  set_tests_properties(adapter_test PROPERTIES TIMEOUT 45)
endif()
ament_package()
